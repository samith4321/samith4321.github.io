Smart Side Door Safety System

An ESP32-based embedded system that prevents unsafe car door openings using PIR and ultrasonic sensors. Built at MIT Manipal · Sep–Oct 2025.

HOW IT WORKS

1

Sensor polling loop runs on ESP32

The microcontroller continuously reads both the PIR and ultrasonic sensors every 100ms using a timed interrupt.

2

PIR detects motion

If the passive infrared sensor picks up movement, the system enters a heightened alert state and begins distance measurement.

3

Ultrasonic measures distance

An HC-SR04 ultrasonic pulse measures exact distance. If object is within the danger threshold (<80cm), a danger flag is set.

4

Alert triggered

The onboard LED flashes and buzzer sounds. The alert persists until the object clears the safe zone.

CORE FIRMWARE (C++)

door_safety.cppESP32 · C++
// Smart Side Door Safety System — ESP32 Firmware
#include <Arduino.h>

// Pin definitions
const int PIR_PIN      = 14;
const int TRIG_PIN     = 12;
const int ECHO_PIN     = 13;
const int LED_PIN      = 2;
const int BUZZER_PIN   = 4;
const int DANGER_CM    = 80;   // danger threshold in cm

long getDistanceCm() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH);
  return duration * 0.034 / 2;
}

void triggerAlert(bool on) {
  digitalWrite(LED_PIN, on ? HIGH : LOW);
  if (on) tone(BUZZER_PIN, 1000);
  else noTone(BUZZER_PIN);
}

void setup() {
  pinMode(PIR_PIN, INPUT);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  Serial.begin(115200);
}

void loop() {
  bool motionDetected = digitalRead(PIR_PIN);

  if (motionDetected) {
    long dist = getDistanceCm();
    Serial.print("Distance: "); Serial.println(dist);

    if (dist < DANGER_CM && dist > 0) {
      triggerAlert(true);   // LED on, buzzer on
    } else {
      triggerAlert(false);  // all clear
    }
  } else {
    triggerAlert(false);
  }
  delay(100);  // poll every 100ms
}
